#!/bin/bash
#SBATCH -N 1 #Number of nodes
#SBATCH -n 16 # number of cores
#SBATCH --mem 64G # memory pool for all cores
#SBATCH --tmp 350G # amount of temporary disk space
#SBATCH -t 2-23:00 # time (D-HH:MM)
#SBATCH -o {0}_flair_diff.%N.%j.log
#SBATCH -e {0}_flair_diff.%N.%j.log
#SBATCH --mail-type=FAIL
#SBATCH --account=def-tetreaum
#SBATCH --mail-user={1}

set -euo pipefail

ml apptainer/1.4.5 bedtools/2.31.0

flair_img="/lustre09/project/6019267/shared/tools/splicing/flair/flair_20260320.sif"
intron_prospector_tool="/lustre09/project/6019267/shared/tools/splicing/flair/intronProspector/bin"
project_path="{0}"
genome_fa="{2}"
annotation_gtf="{3}"
work_dir="{4}"
manifest_combine="$work_dir/manifest_combine_{0}.txt"
manifest_quantify="$work_dir/manifest_quantify_{0}.txt"

mkdir $SLURM_TMPDIR/$project_path
mkdir $SLURM_TMPDIR/$project_path/fastq

# change paths in manifest to paths in SLURM_TMPDIR
sed -i "s#/lustre10/scratch/$USER/#$SLURM_TMPDIR#g" $manifest_quantify
sed -i "s#/lustre10/scratch/$USER/#$SLURM_TMPDIR#g" $manifest_combine

cat $manifest_combine
cat $manifest_quantify

# Clear list
if [[ -f "$SLURM_TMPDIR/$project_path/IPjunctions_{0}_list.txt" ]]; then
    rm $SLURM_TMPDIR/$project_path/IPjunctions_{0}_list.txt
fi

# Find junctions for each samples
for sample in {5}
do
    bam_file=$(grep -P "^${{sample}}\t" {6} | cut -f3)/alignments/${{sample}}_sorted.bam
    echo -e "\nProcessing $bam_file for junctions"

    apptainer run --env PYTHONNOUSERSITE=true \
        -B $SLURM_TMPDIR:$SLURM_TMPDIR \
        $flair_img $intron_prospector_tool/intronProspector \
        -g $genome_fa \
        -b $SLURM_TMPDIR/$project_path/${{sample}}.IPjunctions.bed \
        -C 0.0 \
        -c $SLURM_TMPDIR/$project_path/${{sample}}_intron_calls.tsv \
        $bam_file

    echo -e "$SLURM_TMPDIR/$project_path/${{sample}}_intron_calls.tsv" >> $SLURM_TMPDIR/$project_path/IPjunctions_{0}_list.txt

    # Make fastq files
    echo -e "\nBam to Fastq for $sample"
    bedtools bamtofastq -i $bam_file -fq $SLURM_TMPDIR/$project_path/fastq/${{sample}}.fastq

done

# Merge junctions into one junctions file
echo -e "\nMerging junction files"
apptainer run --env PYTHONNOUSERSITE=true \
    -B $SLURM_TMPDIR:$SLURM_TMPDIR \
    $flair_img $intron_prospector_tool/intronProspectorMerge \
    -i $SLURM_TMPDIR/$project_path/IPjunctions_{0}_list.txt \
    -b $SLURM_TMPDIR/$project_path/IPjunctions_{0}_list_merged.bed

# Generate transcriptomes with FLAIR
for sample in {5}
do
    bam_file=$(grep -P "^$sample\t" {6} | cut -f3)/alignments/${{sample}}_sorted.bam
    echo -e "\nProcessing $bam_file for FLAIR transcriptomes"

    apptainer run --env PYTHONNOUSERSITE=true \
        -B $SLURM_TMPDIR:$SLURM_TMPDIR \
        $flair_img flair transcriptome \
        -g $genome_fa \
        -b $bam_file \
        -f $annotation_gtf \
        --threads 16 \
        --parallelmode bychrom \
        --junction_support 2 \
        --junction_bed $SLURM_TMPDIR/$project_path/IPjunctions_{0}_list_merged.bed \
        -o $SLURM_TMPDIR/$project_path/$sample
done

# Combine transcriptomes
echo -e "\nMerging transcriptomes"
apptainer run --env PYTHONNOUSERSITE=true \
    -B $SLURM_TMPDIR:$SLURM_TMPDIR \
    $flair_img flair combine \
    -m $manifest_combine \
    --include_se \
    -p 5 \
    -f 1 \
    -o $SLURM_TMPDIR/$project_path/{0}_combined

# Quantify splicing
echo -e "\nRunning FLAIR quantify"
apptainer run --env PYTHONNOUSERSITE=true \
    -B $SLURM_TMPDIR:$SLURM_TMPDIR \
    $flair_img flair quantify \
    -r $manifest_quantify \
    -i $SLURM_TMPDIR/$project_path/{0}_combined.fa \
    --threads 16 \
    --temp_dir $SLURM_TMPDIR/$project_path/temp \
    -o $SLURM_TMPDIR/$project_path/{0}_quantify \
    --isoform_bed $SLURM_TMPDIR/$project_path/{0}_combined.bed \
    --stringent \
    --check_splice

# Run differential expression
echo -e "\nRunning FLAIR diff expression"
apptainer run --env PYTHONNOUSERSITE=true \
    -B $SLURM_TMPDIR:$SLURM_TMPDIR \
    $flair_img flair diffexp \
    -q $SLURM_TMPDIR/$project_path/{0}_quantify.counts.tsv \
    --threads 16 \
    --out_dir $SLURM_TMPDIR/$project_path/diffexp \
    --out_dir_force

# Run differential splicing
echo -e "\nRunning FLAIR diff splicing"
apptainer run --env PYTHONNOUSERSITE=true \
    -B $SLURM_TMPDIR:$SLURM_TMPDIR \
    $flair_img flair diffsplice \
    -i $SLURM_TMPDIR/$project_path/{0}_combined.bed \
    -q $SLURM_TMPDIR/$project_path/{0}_quantify.counts.tsv \
    --threads 16 \
    --test \
    --drim1 3 \
    --drim2 3 \
    --drim3 3 \
    --drim4 3 \
    --conditionA {7} \
    --conditionB {8} \
    --out_dir $SLURM_TMPDIR/$project_path/diffsplice \
    --out_dir_force

rsync -azh --no-g --no-p $SLURM_TMPDIR/$project_path $work_dir/

echo -e "\nFLAIR done!"
